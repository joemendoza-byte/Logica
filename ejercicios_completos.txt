3. Fracción mixta a impropia
#include <iostream>
using namespace std;
int main(){
int entero,numerador,denominador;
cout<<"Ingrese la fraccion mixta (entero, numerador/denominador): ";
cin>>entero>>numerador>>denominador;
if(denominador==0){
cout<<"Error: Denominador no puede ser cero."<<endl;
return 0;
}
int numerador_impropio=(entero*denominador)+numerador;
cout<<"Fraccion impropia: "<<numerador_impropio<<"/"<<denominador<<endl;
return 0;
}

8. Tabla de multiplicar rusa
#include <iostream>
using namespace std;
int main(){
int a,b,resultado=0;
cout<<"Ingresa dos numeros a multiplicar: ";
cin>>a>>b;
cout<<endl<<"Tabla de multiplicacion rusa:"<<endl;
cout<<a<<" "<<b<<endl;
while(a>0){
if(a%2==1){
resultado=resultado+b;
cout<<a<<" "<<b<<" <--- Suma "<<b<<endl;
}else{
cout<<a<<" "<<b<<endl;
}
a=a/2;
b=b*2;
}
cout<<" El resultado es: "<<resultado<<endl;
return 0;
}
